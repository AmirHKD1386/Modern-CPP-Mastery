using i64 = long long;
using u64 = unsigned long long;

constexpr i64 INF64 = 1e18;
constexpr int MOD = 1e9 + 7;

inline void optimize_io() noexcept {
    std::ios::sync_with_stdio(false);
    std::cin.tie(nullptr);
}

void solve_case([[maybe_unused]] int test_number) {
    int n = 0;
    if (!(std::cin >> n)) return;

    std::vector<i64> values(n);
    for (auto& val : values) {
        std::cin >> val;
    }

    i64 max_sum = -INF64;
    i64 current_sum = 0;

    for (const auto& val : values) {
        current_sum = std::max(val, current_sum + val);
        max_sum = std::max(max_sum, current_sum);
    }

    std::cout << max_sum << "\n";
}

int main() {
    optimize_io();

    int test_cases = 1;
    // if (std::cin >> test_cases) {}

    for (int t = 1; t <= test_cases; ++t) {
        solve_case(t);
    }

    return 0;
}
